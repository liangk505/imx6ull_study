==================== 一、文件和目录操作 ====================

ls                    # 列出当前目录下的文件和文件夹
ls -l                 # 以长格式列出文件详细信息（权限、大小、时间等）
ls -a                 # 列出所有文件，包括隐藏文件（以.开头的文件）
ls -la                # 长格式列出所有文件（含隐藏文件）
ls -lh                # 以人类可读的大小格式显示（如1K, 234M, 2G）
ls -lt                # 按修改时间排序，最新的在前
ls -ltr               # 按修改时间排序，最旧的在前
ls -R                 # 递归列出所有子目录的内容

pwd                   # 显示当前工作目录的完整路径（Print Working Directory）

cd /path/to/dir       # 切换到指定目录
cd ~                  # 切换到当前用户的家目录
cd -                  # 切换到上一个工作目录
cd ..                 # 切换到上级目录
cd ../..              # 切换到上两级目录

mkdir dirname         # 创建目录（Make Directory）
mkdir -p a/b/c        # 递归创建多级目录（父目录不存在时自动创建）
mkdir -m 755 dirname  # 创建目录并指定权限

rmdir dirname         # 删除空目录
rm -r dirname         # 递归删除目录及其所有内容（危险操作！）
rm -rf dirname        # 强制递归删除，不提示确认（极度危险！）
rm file.txt           # 删除文件
rm -i file.txt        # 删除前逐一确认
rm -f file.txt        # 强制删除文件，不提示

cp source dest        # 复制文件
cp -r src/ dest/      # 递归复制整个目录
cp -i src dest        # 覆盖前询问确认
cp -p src dest        # 保留文件的权限、时间戳等属性
cp -a src/ dest/      # 归档复制（保留所有属性和链接）

mv source dest        # 移动文件或重命名文件/目录
mv -i src dest        # 覆盖前询问确认
mv -f src dest        # 强制移动，不提示

touch filename        # 创建空文件，或更新文件的访问/修改时间戳
touch -t 202512312359 file.txt  # 设置文件的时间戳为指定时间

cat file.txt          # 查看文件全部内容（适合小文件）
cat -n file.txt       # 查看文件内容并显示行号
cat file1 file2 > file3  # 合并多个文件

tac file.txt          # 反向显示文件内容（从最后一行开始）

more file.txt         # 分页查看文件内容（按空格翻页，按q退出）
less file.txt         # 分页查看文件内容（支持上下滚动，更灵活）

head file.txt         # 查看文件前10行
head -n 20 file.txt   # 查看文件前20行
head -c 100 file.txt  # 查看文件前100个字节

tail file.txt         # 查看文件后10行
tail -n 20 file.txt   # 查看文件后20行
tail -f file.txt      # 实时追踪文件新增内容（常用于查看日志）
tail -F file.txt      # 实时追踪文件，文件被删除重建后继续追踪

ln -s /path/to/original linkname  # 创建软链接（符号链接，类似Windows快捷方式）
ln /path/to/original linkname     # 创建硬链接（同一文件的不同入口）

file filename         # 判断文件类型（文本、二进制、图片等）

stat filename         # 显示文件或文件系统的详细状态信息

==================== 二、文件查找和搜索 ====================

find /path -name "*.txt"           # 按文件名查找
find /path -type f -name "*.log"   # 查找指定类型的文件（f=文件, d=目录）
find /path -type d -name "dir*"    # 查找目录
find /path -size +100M             # 查找大于100MB的文件
find /path -mtime -7               # 查找7天内修改过的文件
find /path -mtime +30              # 查找30天前修改过的文件
find /path -atime -1               # 查找1天内访问过的文件
find /path -perm 755               # 查找权限为755的文件
find /path -user username          # 查找属于指定用户的文件
find /path -empty                  # 查找空文件和空目录
find /path -name "*.log" -delete   # 查找并删除匹配的文件
find /path -name "*.log" -exec rm {} \;  # 查找并对每个结果执行命令
find /path -name "*.txt" | xargs grep "pattern"  # 查找文件并用xargs传递结果

locate filename      # 快速查找文件（从数据库中查找，需要先运行updatedb）
locate -i filename   # 不区分大小写查找
updatedb             # 更新locate的文件数据库

which command        # 显示命令的完整路径
whereis command      # 显示命令的二进制文件、源码和man手册页的位置
type command         # 显示命令的类型（内置命令、别名、外部命令等）

==================== 三、文件内容搜索（grep系列）====================

grep "pattern" file.txt            # 在文件中搜索指定模式
grep -i "pattern" file.txt         # 不区分大小写搜索
grep -v "pattern" file.txt         # 反向搜索（显示不匹配的行）
grep -r "pattern" /path/           # 递归搜索目录中的所有文件
grep -n "pattern" file.txt         # 显示匹配行的行号
grep -c "pattern" file.txt         # 统计匹配行数
grep -l "pattern" *.txt            # 只显示包含匹配的文件名
grep -w "word" file.txt            # 精确匹配整个单词
grep -A 3 "pattern" file.txt       # 显示匹配行及其后3行
grep -B 3 "pattern" file.txt       # 显示匹配行及其前3行
grep -C 3 "pattern" file.txt       # 显示匹配行及其前后各3行
grep -E "pattern1|pattern2" file.txt  # 使用扩展正则表达式（等同于egrep）
grep -o "pattern" file.txt         # 只输出匹配的部分

egrep "pattern1|pattern2" file.txt # 扩展grep，支持更丰富的正则表达式
fgrep "literal_string" file.txt    # 快速grep，不解析正则表达式

==================== 四、文件权限管理 ====================

chmod 755 file.txt    # 设置文件权限（rwxr-xr-x）
chmod u+x script.sh   # 给文件所有者添加执行权限
chmod g-w file.txt    # 移除组用户的写权限
chmod o= file.txt     # 清除其他用户的所有权限
chmod -R 755 dir/     # 递归修改目录下所有文件的权限

# 权限数字含义：r=4, w=2, x=1
# 777 = rwxrwxrwx (所有人可读写执行)
# 755 = rwxr-xr-x (所有者全权限，组和其他人可读执行)
# 644 = rw-r--r-- (所有者读写，组和其他人只读)
# 700 = rwx------ (只有所有者有权限)

chown user:group file.txt          # 修改文件的所有者和所属组
chown -R user:group dir/           # 递归修改目录的所有者和所属组
chown user file.txt                # 只修改文件所有者
chown :group file.txt              # 只修改文件所属组

chgrp groupname file.txt           # 修改文件所属组
chgrp -R groupname dir/            # 递归修改目录所属组

umask 022             # 设置默认文件权限掩码（新文件默认权限=666-umask）
umask                 # 查看当前umask值

usermod -U username            # 解锁用户账户

passwd                # 修改当前用户密码
passwd username       # 修改指定用户密码（需要root权限）
passwd -l username    # 锁定用户密码
passwd -u username    # 解锁用户密码

groupadd groupname    # 创建新组
groupdel groupname    # 删除组

==================== 六、进程管理 ====================

ps                    # 显示当前shell的进程
ps aux                # 显示系统中所有进程的详细信息
ps -ef                # 显示所有进程的完整信息
ps -eo pid,ppid,cmd,%mem,%cpu  # 自定义显示进程信息

top                   # 实时显示进程动态（按q退出，按k输入PID杀进程）
htop                  # top的增强版（界面更友好，需单独安装）

pgrep process_name    # 根据进程名查找PID
pgrep -l process_name # 查找并显示进程名和PID
pgrep -u username     # 查找指定用户的进程

pidof process_name    # 查找进程的PID

kill PID              # 终止指定PID的进程（发送TERM信号）
kill -9 PID           # 强制终止进程（发送KILL信号，不可被忽略）
kill -15 PID          # 正常终止进程（默认信号）
kill -l               # 列出所有信号

killall process_name  # 根据进程名终止所有匹配的进程
killall -9 pname      # 强制终止所有匹配进程

pkill process_name    # 根据进程名终止进程（支持模式匹配）
pkill -f pattern      # 根据完整命令行匹配并终止进程

bg                    # 将暂停的后台任务放到后台继续运行
fg                    # 将后台任务调到前台运行
jobs                  # 显示当前shell的后台任务列表

nohup command &       # 后台运行命令，不受终端关闭影响
command &             # 将命令放到后台运行

nice -n 10 command    # 以较低优先级运行命令（-20最高，19最低）
renice -n 5 -p PID    # 修改运行中进程的优先级

==================== 七、磁盘和存储管理 ====================

df -h                 # 以人类可读格式显示磁盘分区使用情况
df -i                 # 显示inode使用情况
df -T                 # 显示文件系统类型

du -sh /path/dir/     # 显示目录的总大小
du -h --max-depth=1   # 显示当前目录下各子目录的大小
du -sh *              # 显示当前目录下每个文件/目录的大小
du -ah                # 显示所有文件和目录的大小

lsblk                 # 列出所有块设备（磁盘、分区等）

dd if=/dev/sda of=/backup/disk.img bs=4M status=progress  # 磁盘/分区备份
dd if=input.iso of=/dev/sdb bs=4M status=progress         # 写入ISO到USB

==================== 八、压缩和解压缩 ====================

# --- tar ---
tar -cvf archive.tar dir/           # 创建tar归档（c=创建, v=详细, f=文件）
tar -xvf archive.tar                # 解压tar归档
tar -tvf archive.tar                # 查看tar归档内容不解压
tar -czvf archive.tar.gz dir/       # 创建gzip压缩的tar归档
tar -xzvf archive.tar.gz            # 解压.tar.gz文件
tar -cjvf archive.tar.bz2 dir/      # 创建bzip2压缩的tar归档
tar -xjvf archive.tar.bz2           # 解压.tar.bz2文件
tar -cJvf archive.tar.xz dir/       # 创建xz压缩的tar归档
tar -xJvf archive.tar.xz            # 解压.tar.xz文件

# --- gzip ---
gzip file.txt         # 压缩文件（生成file.txt.gz）
gzip -d file.txt.gz   # 解压.gz文件
gunzip file.txt.gz    # 解压.gz文件（同上）
gzip -k file.txt      # 压缩但保留原文件

# --- zip/unzip ---
zip archive.zip file1 file2         # 创建zip压缩包
zip -r archive.zip dir/             # 递归压缩目录
unzip archive.zip                   # 解压zip文件
unzip -l archive.zip                # 查看zip内容不解压

# --- 其他 ---
bzip2 file.txt        # bzip2压缩（生成.bz2文件，压缩率比gzip高）
bunzip2 file.txt.bz2  # 解压.bz2文件
xz file.txt           # xz压缩（高压缩率，生成.xz文件）
unxz file.txt.xz      # 解压.xz文件

==================== 十一、软件包管理 ====================

==================== 十三、文本处理 ====================

sort file.txt         # 按行排序文件内容

sort -n file.txt      # 按数字排序
sort -u file.txt      # 排序并去重
sort -k 2 file.txt    # 按第2列排序
sort -t ':' -k 3 file.txt  # 按冒号分隔的第3列排序

uniq file.txt         # 去除重复行（需先排序）
uniq -c file.txt      # 统计每行重复次数
uniq -d file.txt      # 只显示重复的行
uniq -u file.txt      # 只显示不重复的行

wc file.txt           # 统计行数、单词数、字节数
wc -l file.txt        # 只统计行数
wc -w file.txt        # 只统计单词数
wc -c file.txt        # 只统计字节数
wc -m file.txt        # 只统计字符数

cut -d':' -f1 /etc/passwd   # 按冒号分隔，取第1列
cut -d',' -f1,3 file.csv    # 取第1和第3列
cut -c1-5 file.txt          # 取每行的第1到5个字符

tr 'a-z' 'A-Z' < file.txt   # 小写转大写
tr -d '\r' < file.txt       # 删除回车符
echo "a b c" | tr ' ' ','   # 替换空格为逗号

sed 's/old/new/' file.txt   # 替换每行第一个匹配
sed 's/old/new/g' file.txt  # 替换每行所有匹配
sed 's/old/new/gI' file.txt # 不区分大小写替换
sed -i 's/old/new/g' file.txt  # 直接在文件中替换
sed '/pattern/d' file.txt   # 删除匹配的行
sed -n '5,10p' file.txt     # 打印第5到10行
sed '2a\new line' file.txt  # 在第2行后插入

awk '{print $1}' file.txt   # 打印第1列
awk '{print $1,$3}' file.txt  # 打印第1和第3列
awk -F':' '{print $1}' file.txt  # 指定分隔符
awk '{sum+=$1} END {print sum}' file.txt  # 计算第1列和
awk '/pattern/' file.txt    # 打印匹配的行
awk 'NR==5' file.txt        # 打印第5行
awk 'NF>5' file.txt         # 打印字段数大于5的行

diff file1.txt file2.txt    # 比较两个文件的差异
diff -u file1 file2         # 统一格式显示差异（类似git diff）
diff -r dir1/ dir2/         # 递归比较两个目录

tee file.txt          # 从标准输入读取并同时写入文件和标准输出

column -t file.txt    # 将内容格式化为表格对齐

==================== 十四、I/O 重定向和管道 ====================

command 2>&1                # 将标准错误重定向到标准输出
command &> all.txt          # 将标准输出和错误都重定向到文件
command < file.txt          # 将文件作为标准输入

command1 | command2         # 管道：将command1的输出作为command2的输入
command1 | tee file.txt     # 输出到文件和管道
command > /dev/null 2>&1    # 静默执行，丢弃所有输出

    # 条件成立时执行

    # 另一条件成立时执行

  11. 使用 || 可串联命令（前一个失败才执行后一个）
